##### leak_stress
export A=1
cat << END > f.txt
one $A
two
END
cat f.txt | tr a-z A-Z | wc -l
cat < f.txt >> g.txt
nonexistent_cmd_xyz
echo $?
echo hi > h.txt
cat h.txt g.txt
cat << 'END' | cat | wc -l
a
b
END
rm -f f.txt g.txt h.txt

##### leak_stress_repeat

cat << END | tr a-z A-Z | cat > out
x
y
END
cat out
nonexistent_zzz
cat < /no/such
echo done
rm -f f1 f2 out
